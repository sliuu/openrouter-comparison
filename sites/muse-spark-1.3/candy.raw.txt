function changeQty(id,d){

function cartCountTotal(){let c=0;for(let k in cart)c+=cart[k];return c}
function cartSubtotal(){
 let s=0;for(let k in cart){
  if(k==='box') s+=24*cart[k];
  else{let p=PRODUCTS.find(x=>x.id==k); if(p) s+=p.price*cart[k];}
  // build box item
  if(String(k).startsWith('build-')) s+=21*cart[k];
 }
 return s;
}
