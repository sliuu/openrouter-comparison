  var lenis = null;

    gsap.ticker.add(function(t){ lenis.raf(t * 1000); });
    gsap.ticker.lagSmoothing(0);
    lenis.stop();
  }
  function scrollToTarget(sel){
    var el = document.querySelector(sel);
    if (!el) return;
    if (lenis) lenis.scrollTo(el, { offset: -70, duration: 1.4 });
    else el.scrollIntoView({ behavior: reduced ? 'auto' : 'smooth' });
  }

  /* ---------- anchor links ---------- */
  document.querySelectorAll('a[href^="#"]').forEach(function(a){

      var href = a.getAttribute('href');
      if (href.length > 1) { e.preventDefault(); closeMenu(); scrollToTarget(href); }
    });
  });
  document.getElementById('to-top').addEventListener('click', function(){
    if (lenis) lenis.scrollTo(0, { duration: 1.4 }); else window.scrollTo({top:0});
  });

  /* ---------- custom cursor ---------- */
  if (finePointer && !reduced) {
    document.documentElement.classList.add('has-cursor');
    var dot = document.getElementById('cursor-dot');
    var ring = document.getElementById('cursor-ring');
    var label = document.getElementById('cursor-label');
    var mx = innerWidth/2, my = innerHeight/2, rx = mx, ry = my;
    window.addEventListener('mousemove', function(e){
      mx = e.clientX; my = e.clientY;
      dot.style.left = mx + 'px'; dot.style.top = my + 'px';
    });
    gsap.ticker.add(function(){
      rx += (mx - rx) * 0.14; ry += (my - ry) * 0.14;
      ring.style.left = rx + 'px'; ring.style.top = ry + 'px';
    });
    document.querySelectorAll('a, button, input, textarea, select, .faq-btn').forEach(function(el){

  if (finePointer && !reduced) {
    document.querySelectorAll('[data-magnetic]').forEach(function(btn){
      btn.addEventListener('mousemove', function(e){
        var r = btn.getBoundingClientRect();
        gsap.to(btn, { x: (e.clientX - r.left - r.width/2) * 0.28, y: (e.clientY - r.top - r.height/2) * 0.28, duration: 0.4, ease: 'power2.out' });
      });
      btn.addEventListener('mouseleave', function(){
        gsap.to(btn, { x: 0, y: 0, duration: 0.6, ease: 'elastic.out(1, 0.4)' });
      });
    });
  }

  /* ---------- nav behavior ---------- */
  var nav = document.getElementById('nav');
  var lastY = 0;
  function onScrollPos(y){
    nav.classList.toggle('nav-scrolled', y > 60);
    if (y > 420 && y > lastY + 4) nav.classList.add('nav-hidden');
    else if (y < lastY - 4) nav.classList.remove('nav-hidden');
    lastY = y;
  }
  if (lenis) lenis.on('scroll', function(e){ onScrollPos(e.scroll); });
  else window.addEventListener('scroll', function(){ onScrollPos(window.scrollY); }, { passive: true });

  /* ---------- menu overlay ---------- */
  var overlay = document.getElementById('menu-overlay');
  var menuTl = gsap.timeline({ paused: true })
    .to(overlay, { autoAlpha: 1, duration: 0.4, ease: 'power2.out' })
    .fromTo('.menu-link', { y: 44, autoAlpha: 0 }, { y: 0, autoAlpha: 1, stagger: 0.06, duration: 0.55, ease: 'power3.out' }, '-=0.1');
  var menuOpen = false;
  document.getElementById('menu-open').addEventListener('click', function(){
    menuOpen = true; menuTl.timeScale(1).play(); if (lenis) lenis.stop();
  });
  function closeMenu(){
    if (!menuOpen) return;
    menuOpen = false; menuTl.timeScale(1.6).reverse(); if (lenis) lenis.start();
  }
  document.getElementById('menu-close').addEventListener('click', closeMenu);
  window.addEventListener('keydown', function(e){ if (e.key === 'Escape') closeMenu(); });

  /* ---------- preloader + hero intro ---------- */
  var heroLines = document.querySelectorAll('.hero-line');
  var heroBits = ['#hero-eyebrow', '#hero-sub', '#hero-ctas', '#hero-meta', '#hero-badge'];

  if (reduced) {
    preloader.style.display = 'none';
    if (lenis) lenis.start();
  } else {
    gsap.set(heroLines, { yPercent: 120 });
    gsap.set(heroBits.map(function(s){ return s; }).join(','), { autoAlpha: 0, y: 26 });
    gsap.set('#pre-mark', { yPercent: 110 });
    gsap.set('#pre-sub', { autoAlpha: 0 });

    var counter = { v: 0 };
    var countEl = document.getElementById('pre-count');
    var preTl = gsap.timeline();
    preTl
      .to('#pre-mark', { yPercent: 0, duration: 0.9, ease: 'power4.out' }, 0.15)
      .to('#pre-sub', { autoAlpha: 1, duration: 0.6 }, 0.6)
      .to(counter, { v: 100, duration: 1.5, ease: 'power2.inOut',
        onUpdate: function(){ countEl.textContent = Math.round(counter.v); } }, 0.2)
      .to(preloader, { yPercent: -100, duration: 0.9, ease: 'power4.inOut' }, '+=0.15')
      .set(preloader, { display: 'none' })
      .add(function(){ if (lenis) lenis.start(); }, '-=0.5')
      .to(heroLines, { yPercent: 0, duration: 1.15, ease: 'power4.out', stagger: 0.1 }, '-=0.55')
      .to(heroBits.join(','), { autoAlpha: 1, y: 0, duration: 0.9, ease: 'power3.out', stagger: 0.08 }, '-=0.7');
  }

  /* ---------- hero parallax ---------- */
  if (!reduced) {
    gsap.to('#hero-bg', { yPercent: 16, ease: 'none',
      scrollTrigger: { trigger: '#hero', start: 'top top', end: 'bottom top', scrub: true } });
    gsap.to('#footer-word', { yPercent: -12, ease: 'none',
      scrollTrigger: { trigger: '#footer-word', start: 'top bottom', end: 'bottom bottom', scrub: true } });
  }

  /* ---------- scroll reveals ---------- */
  if (!reduced) {
    gsap.utils.toArray('[data-reveal]').forEach(function(el){
      gsap.fromTo(el, { y: 44, autoAlpha: 0 },
        { y: 0, autoAlpha: 1, duration: 1, ease: 'power3.out',
          scrollTrigger: { trigger: el, start: 'top 87%' } });
    });
    gsap.utils.toArray('[data-lines]').forEach(function(block){
      var lines = block.querySelectorAll('.rl');
      gsap.fromTo(lines, { yPercent: 120 },
        { yPercent: 0, duration: 1.1, ease: 'power4.out', stagger: 0.09,
          scrollTrigger: { trigger: block, start: 'top 86%' } });
    });
  }

  /* ---------- counters ---------- */
  document.querySelectorAll('[data-count]').forEach(function(el){
    var end = parseFloat(el.dataset.count);
    var useComma = el.dataset.comma === '1';
    function fmt(v){ var n = Math.round(v); return useComma ? n.toLocaleString('en-US') : n; }
    if (reduced) { el.textContent = fmt(end); return; }
    var obj = { v: 0 };
    ScrollTrigger.create({
      trigger: el, start: 'top 88%', once: true,
      onEnter: function(){
        gsap.to(obj, { v: end, duration: 2.1, ease: 'power2.out',
          onUpdate: function(){ el.textContent = fmt(obj.v); } });
      }
    });
  });

  /* ---------- practice hover image ---------- */
  var pSection = document.getElementById('practice');
  var pFloat = document.getElementById('practice-float');
  var pImg = document.getElementById('practice-float-img');
  var rows = document.querySelectorAll('.practice-row');

  rows.forEach(function(r){ var i = new Image(); i.src = r.dataset.img; }); // preload

  if (finePointer && !reduced && pFloat) {
    gsap.set(pFloat, { xPercent: -50, yPercent: -50, autoAlpha: 0, scale: 0.88, rotate: -2 });
    var xTo = gsap.quickTo(pFloat, 'x', { duration: 0.55, ease: 'power3' });
    var yTo = gsap.quickTo(pFloat, 'y', { duration: 0.55, ease: 'power3' });
    pSection.addEventListener('mousemove', function(e){ xTo(e.clientX); yTo(e.clientY); });
    rows.forEach(function(row){
      row.addEventListener('mouseenter', function(){
        pImg.src = row.dataset.img;
        gsap.to(pFloat, { autoAlpha: 1, scale: 1, rotate: 0, duration: 0.45, ease: 'power3.out' });
        gsap.fromTo(pImg, { scale: 1.25 }, { scale: 1, duration: 0.7, ease: 'power3.out' });
      });
    });
    pSection.addEventListener('mouseleave', function(){
      gsap.to(pFloat, { autoAlpha: 0, scale: 0.88, rotate: -2, duration: 0.35, ease: 'power2.in' });
    });
  }

  /* ---------- testimonial slider ---------- */
  var slides = Array.prototype.slice.call(document.querySelectorAll('.t-slide'));
  var tIndex = document.getElementById('t-index');
  var ti = 0, tTimer = null;
  slides.forEach(function(s, i){ gsap.set(s, { autoAlpha: i === 0 ? 1 : 0, y: i === 0 ? 0 : 24 }); });

  function tGo(n){
    var next = (n + slides.length) % slides.length;
    if (next === ti) return;
    gsap.to(slides[ti], { autoAlpha: 0, y: -24, duration: 0.5, ease: 'power2.in' });
    gsap.fromTo(slides[next], { autoAlpha: 0, y: 30 }, { autoAlpha: 1, y: 0, duration: 0.7, ease: 'power3.out', delay: 0.35 });
    ti = next;
    tIndex.textContent = ('0' + (ti + 1)).slice(-2);
  }
  function tAuto(){ clearInterval(tTimer); tTimer = setInterval(function(){ tGo(ti + 1); }, 6500); }
  document.getElementById('t-next').addEventListener('click', function(){ tGo(ti + 1); tAuto(); });
  document.getElementById('t-prev').addEventListener('click', function(){ tGo(ti - 1); tAuto(); });
  document.getElementById('t-stage').addEventListener('mouseenter', function(){ clearInterval(tTimer); });
  document.getElementById('t-stage').addEventListener('mouseleave', tAuto);
  tAuto();

  /* ---------- FAQ accordion ---------- */
  var faqItems = document.querySelectorAll('.faq-item');
  faqItems.forEach(function(item){
    var btn = item.querySelector('.faq-btn');
    var panel = item.querySelector('.faq-panel');
    btn.addEventListener('click', function(){
      var isOpen = item.classList.contains('open');
      faqItems.forEach(function(o){
        o.classList.remove('open');
        o.querySelector('.faq-panel').style.height = '0px';
        o.querySelector('.faq-btn').setAttribute('aria-expanded', 'false');
      });
      if (!isOpen) {
        item.classList.add('open');
        panel.style.height = panel.scrollHeight + 'px';
        btn.setAttribute('aria-expanded', 'true');
      }
    });
  });
  window.addEventListener('resize', function(){
    var open = document.querySelector('.faq-item.open .faq-panel');
    if (open) open.style.height = open.scrollHeight + 'px';
  });

  /* ---------- contact form ---------- */
  var form = document.getElementById('case-form');
  var formError = document.getElementById('form-error');
  var formSuccess = document.getElementById('form-success');
  form.addEventListener('submit', function(e){
    e.preventDefault();
    var required = ['f-name', 'f-phone', 'f-type', 'f-msg'];
    var valid = true;
    required.forEach(function(id){
      var f = document.getElementById(id);
      var empty = !f.value.trim();
      f.style.borderColor = empty ? '#c9a25e' : '';
      if (empty) valid = false;
    });
    if (!valid) {
      formError.classList.remove('hidden');
      gsap.fromTo(formError, { x: -6 }, { x: 0, duration: 0.4, ease: 'elastic.out(1,0.3)' });
      return;
    }
    formError.classList.add('hidden');
    gsap.to(form, { autoAlpha: 0, y: -14, duration: 0.45, ease: 'power2.in', onComplete: function(){
      form.style.visibility = 'hidden';
      formSuccess.classList.remove('hidden');
      gsap.fromTo(formSuccess, { autoAlpha: 0, y: 24 }, { autoAlpha: 1, y: 0, duration: 0.7, ease: 'power3.out' });
    }});
  });

  /* ---------- footer year ---------- */
  document.getElementById('year').textContent = new Date().getFullYear();

})();
</script>
