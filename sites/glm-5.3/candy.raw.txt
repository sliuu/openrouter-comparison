/* ---------- lollipop spin ---------- */
(function(){
  const lolli = $('#lolli'), spinG = $('#lolliSpin'), spiral = $('#spiral');
  // build the swirl path
  let d=''; const turns=3, r0=8, r1=100, total=turns*Math.PI*2;
  for(let a=0;a<=total;a+=0.12){
    const r = r0 + (r1-r0)*a/total;
    d += (a===0?'M':'L') + (160 + r*Math.cos(a)).toFixed(1) + ' ' + (150 + r*Math.sin(a)).toFixed(1) + ' ';
  }
  spiral.setAttribute('d', d);

  let rot=0, vel=0, dragging=false, lastA=0, lastT=0;
  function center(){
    const r = lolli.getBoundingClientRect();
    return {x: r.left + r.width*(160/320), y: r.top + r.height*(150/470)};
  }
  lolli.addEventListener('pointerdown', e=>{
    e.preventDefault();
    dragging = true; vel = 0;
    try{ lolli.setPointerCapture(e.pointerId); }catch(err){}
    const c = center();
    lastA = Math.atan2(e.clientY-c.y, e.clientX-c.x);
    lastT = performance.now();
    lolli.classList.add('grabbing');
  });
  lolli.addEventListener('pointermove', e=>{
    if(!dragging) return;
    const c = center();
    const a = Math.atan2(e.clientY-c.y, e.clientX-c.x);
    let delta = a - lastA;
    while(delta >  Math.PI) delta -= Math.PI*2;
    while(delta < -Math.PI) delta += Math.PI*2;
    const now = performance.now(); const dt = Math.max(8, now-lastT);
    rot += delta*180/Math.PI;
    vel = 0.75*vel + 0.25*(delta*180/Math.PI)*(16/dt);
    lastA = a; lastT = now;
  });
  function endDrag(){ dragging = false; lolli.classList.remove('grabbing'); }
  lolli.addEventListener('pointerup', endDrag);
  lolli.addEventListener('pointercancel', endDrag);

  (function loop(){
    if(!dragging){
      rot += vel; vel *= 0.965;
      if(!RM) rot += 0.05; // lazy idle turn
    }
    spinG.setAttribute('transform', 'rotate('+rot.toFixed(2)+' 160 150)');
    requestAnimationFrame(loop);
  })();
})();

function slotPos(i){
  const ROWS = [6,5,5,4,4,3,2];
  const W = jarEl.clientWidth || 300, H = jarEl.clientHeight || 330;
  let idx=0, row=0;
  while(idx + ROWS[row] <= i){ idx += ROWS[row]; row++; }
  const k = i-idx, n = ROWS[row];
  const m = 22;
  const x = m + (W-2*m)*(k+0.5)/n + rnd(-5,5);
  const y = H - 24 - row*26 + rnd(-3,3);
  return { x: Math.min(W-20, Math.max(20, x)), y: Math.max(26, y) };
}
