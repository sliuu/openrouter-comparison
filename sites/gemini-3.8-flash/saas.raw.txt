    // ROI CALCULATOR
    function calculateROI() {
      const teamSize = parseInt(document.getElementById('team-size').value);
      const campaigns = parseInt(document.getElementById('campaigns-count').value);
      const agency = parseInt(document.getElementById('agency-spend').value);

      document.getElementById('team-size-val').innerText = teamSize + " members";
      document.getElementById('campaigns-val').innerText = campaigns + " campaigns";
      document.getElementById('agency-val').innerText = "$" + agency.toLocaleString() + " / mo";

      // Simple modeling logic
      // Hours per campaign manually: ~70 hours across copy, visual, QA, tagging
      // With Connected: ~6 hours
      const hoursSavedPerQuarter = campaigns * 48; 
      // Financial savings: 75% of agency spend + team efficiency
      const annualAgencySavings = agency * 12 * 0.70;
      const internalTeamTimeValuation = hoursSavedPerQuarter * 4 * 75; // $75/hr assumed
      const totalSavings = Math.round(annualAgencySavings + internalTeamTimeValuation);

      document.getElementById('cost-savings').innerText = "$" + totalSavings.toLocaleString();
      document.getElementById('hours-saved').innerText = hoursSavedPerQuarter.toLocaleString() + " hrs";
    }
