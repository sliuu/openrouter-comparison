    // 3. CART SYSTEM
    function addToCart(productId, customBundle = null) {
      if (customBundle) {
        cart.push(customBundle);
        showToast("Tuck Box Added!", "Your custom 4-pack bundle was added to the haul.");
      } else {
        const product = PRODUCTS.find(p => p.id === productId);
        if (!product) return;

        const existing = cart.find(item => item.id === productId && !item.isBundle);
        if (existing) {
          existing.qty += 1;
        } else {
          cart.push({
            id: product.id,
            name: product.name,
            price: product.price,
            emoji: product.emoji,
            qty: 1,
            isBundle: false
          });
        }
        showToast("Tucked in your bag!", `${product.name} added.`);
      }

      updateCartUI();
      openCart();
    }

    function updateCartUI() {
      const cartCountEl = document.getElementById('cartCount');
      const totalCount = cart.reduce((sum, item) => sum + (item.qty || 1), 0);
      cartCountEl.textContent = totalCount;

      // Render items
      const listEl = document.getElementById('cartItemList');
      listEl.innerHTML = '';

      if (cart.length === 0) {
        listEl.innerHTML = `
          <div class="h-64 flex flex-col items-center justify-center text-center p-6 text-muted">
            <span class="text-5xl mb-3">🧺</span>
            <p class="font-serif font-bold text-lg text-ink">Your summer tote is empty.</p>
            <p class="text-xs text-muted max-w-xs mt-1">Pick some sunny drops or custom mix your own 4-flavor tuck box above!</p>
          </div>
        `;
      } else {
        cart.forEach((item, index) => {
          const itemEl = document.createElement('div');
          itemEl.className = "p-3.5 bg-white rounded-2xl border-2 border-ink flex items-center justify-between gap-3 shadow-pop-sm";
          
          if (item.isBundle) {
            itemEl.innerHTML = `
              <div class="flex items-center gap-3">
                <div class="w-12 h-12 bg-mintLight rounded-xl border border-ink flex items-center justify-center text-xl shrink-0">
                  🎁
                </div>
                <div>
                  <h4 class="font-serif font-black text-sm text-ink leading-tight">${item.name}</h4>
                  <p class="text-[10px] text-muted font-bold truncate max-w-[170px]">${item.items.join(', ')}</p>
                  <span class="font-bold text-xs text-strawberry">$${item.price.toFixed(2)}</span>
                </div>
              </div>
              <button onclick="removeCartItem(${index})" class="text-muted hover:text-strawberry p-1" title="Remove bundle">
                <i data-lucide="trash-2" class="w-4 h-4"></i>
              </button>
            `;
          } else {
            itemEl.innerHTML = `
              <div class="flex items-center gap-3">
                <div class="w-12 h-12 bg-blush rounded-xl border border-ink flex items-center justify-center text-2xl shrink-0">
                  ${item.emoji}
                </div>
                <div>
                  <h4 class="font-serif font-bold text-sm text-ink leading-tight">${item.name}</h4>
                  <div class="flex items-center gap-2 mt-1">
                    <span class="font-bold text-xs text-ink">$${(item.price * item.qty).toFixed(2)}</span>
                    <span class="text-[10px] text-muted">($${item.price.toFixed(2)} ea)</span>
                  </div>
                </div>
              </div>
              <div class="flex items-center gap-2">
                <div class="flex items-center border border-ink rounded-lg bg-cream overflow-hidden">
                  <button onclick="changeQty(${index}, -1)" class="px-2 py-0.5 hover:bg-blush font-bold text-xs">-</button>
                  <span class="px-2 font-black text-xs">${item.qty}</span>
                  <button onclick="changeQty(${index}, 1)" class="px-2 py-0.5 hover:bg-blush font-bold text-xs">+</button>
                </div>
                <button onclick="removeCartItem(${index})" class="text-muted hover:text-strawberry p-1">
                  <i data-lucide="trash-2" class="w-4 h-4"></i>
                </button>
              </div>
            `;
          }
          listEl.appendChild(itemEl);
        });
      }

      // Calculate totals
      const subtotal = cart.reduce((sum, item) => sum + (item.price * (item.qty || 1)), 0);
      const isFreeShipping = subtotal >= FREE_SHIPPING_THRESHOLD || subtotal === 0;
      const shippingCost = isFreeShipping ? 0 : SHIPPING_COST;
      const total = subtotal + shippingCost;

      document.getElementById('cartSubtotal').textContent = `$${subtotal.toFixed(2)}`;
      document.getElementById('cartShippingText').textContent = isFreeShipping ? 'FREE' : `$${SHIPPING_COST.toFixed(2)}`;
      document.getElementById('cartTotal').textContent = `$${total.toFixed(2)}`;

      // Free shipping progress bar
      const progress = Math.min(100, (subtotal / FREE_SHIPPING_THRESHOLD) * 100);
      document.getElementById('shippingBar').style.width = `${progress}%`;
      document.getElementById('shippingGoalPercent').textContent = `${Math.round(progress)}%`;

      const diff = FREE_SHIPPING_THRESHOLD - subtotal;
      if (diff > 0 && subtotal > 0) {
        document.getElementById('shippingGoalText').textContent = `Add $${diff.toFixed(2)} more for Free Shipping!`;
      } else if (subtotal >= FREE_SHIPPING_THRESHOLD) {
        document.getElementById('shippingGoalText').textContent = `🎉 You unlocked Free Chilled Shipping!`;
      } else {
        document.getElementById('shippingGoalText').textContent = `Add $${FREE_SHIPPING_THRESHOLD.toFixed(2)} for Free Shipping!`;
      }

      lucide.createIcons();
    }

    function changeQty(index, change) {
      if (!cart[index]) return;
      cart[index].qty += change;
      if (cart[index].qty <= 0) {
        cart.splice(index, 1);
      }
      updateCartUI();
    }

    function removeCartItem(index) {
      cart.splice(index, 1);
      updateCartUI();
    }

      window.closeCart = close;

      cartBtn.addEventListener('click', open);
      closeBtn.addEventListener('click', close);
      backdrop.addEventListener('click', close);

      checkoutBtn.addEventListener('click', () => {
        if (cart.length === 0) {

    function setupBoxBuilder() {
      const optionsContainer = document.getElementById('boxFlavorOptions');
      const addBoxBtn = document.getElementById('addBoxToCartBtn');

      function updateBoxUI() {
        // Update slots
        for (let i = 0; i < 4; i++) {
          const slot = document.getElementById(`slot-${i}`);
          if (customBox[i]) {
            slot.className = "aspect-square border-2 border-ink rounded-2xl flex flex-col items-center justify-center p-2 text-center bg-sunshine/30 shadow-pop-sm animate-in zoom-in-75 relative group";
            slot.innerHTML = `
              <button onclick="removeBoxSlot(${i})" class="absolute -top-1 -right-1 w-5 h-5 bg-strawberry text-white rounded-full flex items-center justify-center text-[10px] border border-ink opacity-90 hover:opacity-100" title="Remove">✕</button>
              <span class="text-2xl">${customBox[i].emoji}</span>
              <span class="text-[10px] font-bold text-ink mt-1 truncate w-full">${customBox[i].name.split(' ')[0]}</span>
            `;
          } else {
            slot.className = "aspect-square border-2 border-dashed border-ink/40 rounded-2xl flex flex-col items-center justify-center p-2 text-center bg-cream transition-all";
            slot.innerHTML = `
              <span class="text-2xl opacity-40">📦</span>
              <span class="text-[10px] font-bold text-muted mt-1">Empty Slot</span>
            `;
          }
        }

        countDisplay.textContent = `${customBox.length} / 4 Selected`;
        progressBar.style.width = `${(customBox.length / 4) * 100}%`;

        if (customBox.length === 4) {
          addBoxBtn.disabled = false;
          addBoxBtn.classList.remove('opacity-40', 'cursor-not-allowed', 'bg-ink');
          addBoxBtn.classList.add('bg-strawberry', 'hover:bg-tangerine', 'text-white', 'shadow-pop');
          addBoxBtn.textContent = "Add 4-Pack to Cart ($22)";
        } else {
          addBoxBtn.disabled = true;
          addBoxBtn.classList.add('opacity-40', 'cursor-not-allowed', 'bg-ink');
          addBoxBtn.classList.remove('bg-strawberry', 'hover:bg-tangerine', 'text-white', 'shadow-pop');
          addBoxBtn.textContent = `Pick ${4 - customBox.length} More Pouches`;
        }
      }

      window.removeBoxSlot = function(index) {
        customBox.splice(index, 1);
        updateBoxUI();
      };

      resetBtn.addEventListener('click', () => {
        customBox = [];
        updateBoxUI();
      });

      addBoxBtn.addEventListener('click', () => {
        if (customBox.length === 4) {
          const bundleItem = {
            id: 'custom-tuck-box-' + Date.now(),
            name: 'Custom Summer Tuck Box (4-Pack)',
            price: 22.00,
            isBundle: true,
            qty: 1,
            items: customBox.map(b => b.name)
          };
          addToCart(null, bundleItem);
          customBox = [];
          updateBoxUI();
        }
      });

      updateBoxUI();
    }
